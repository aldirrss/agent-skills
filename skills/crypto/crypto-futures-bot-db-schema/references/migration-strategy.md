## Alembic setup with SQLModel

```bash
pip install alembic asyncpg sqlmodel

# Initialize (run once)
alembic init -t async alembic
```

```python
# alembic/env.py — replace default content with this
from logging.config import fileConfig
from sqlalchemy.ext.asyncio import create_async_engine
from sqlalchemy import pool
from alembic import context
from sqlmodel import SQLModel

# Import ALL models so SQLModel.metadata knows about every table
from db.models.exchange   import Exchange, Account, Symbol
from db.models.worker     import Worker, BotSession
from db.models.signal     import Signal
from db.models.order      import Order
from db.models.trade      import Trade, FundingPayment
from db.models.snapshot   import PnlSnapshot

config  = context.config
fileConfig(config.config_file_name)

target_metadata = SQLModel.metadata   # ← key line: use SQLModel metadata

def run_migrations_offline():
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        compare_type=True,      # detect column type changes
    )
    with context.begin_transaction():
        context.run_migrations()

async def run_migrations_online():
    connectable = create_async_engine(
        config.get_main_option("sqlalchemy.url"),
        poolclass=pool.NullPool,   # no pool for migrations
    )
    async with connectable.connect() as conn:
        await conn.run_sync(do_run_migrations)
    await connectable.dispose()

def do_run_migrations(conn):
    context.configure(
        connection=conn,
        target_metadata=target_metadata,
        compare_type=True,
    )
    with context.begin_transaction():
        context.run_migrations()

import asyncio
if context.is_offline_mode():
    run_migrations_offline()
else:
    asyncio.run(run_migrations_online())
```

## Naming convention

```
{NNN}_{verb}_{object}.py

NNN:   zero-padded 3-digit sequence (001, 002, 003)
verb:  initial / add / remove / rename / alter / create / drop
object: what changed (snake_case)

Examples:
  001_initial_schema.py
  002_add_llm_score_to_signals.py
  003_add_index_trades_strategy.py
  004_rename_pnl_to_net_pnl.py
  005_add_bitget_exchange.py
```

**Never reuse sequence numbers.** If you delete a migration, leave the gap — the numbers are for humans, not machines.

## Generating migrations

```bash
# Auto-generate from model changes (always review before applying)
alembic revision --autogenerate -m "add_llm_score_to_signals"

# Create blank migration (for data migrations or index-only changes)
alembic revision -m "add_index_trades_strategy"

# Apply pending migrations
alembic upgrade head

# Apply one step at a time
alembic upgrade +1

# Check current state
alembic current

# See migration history
alembic history --verbose
```

**Always review autogenerated migrations before applying.** Alembic sometimes misses:
- Enum type changes (PostgreSQL enums need manual `ALTER TYPE ... ADD VALUE`)
- BRIN vs BTREE index differences
- `server_default` changes
- Complex constraints

## Safe migration workflow

```
1. Write model change in Python (SQLModel class)
         ↓
2. alembic revision --autogenerate -m "description"
         ↓
3. Review the generated migration file:
   - Does upgrade() do what you intended?
   - Does downgrade() correctly undo it?
   - Are there any data-destructive operations?
         ↓
4. Test on dev database first:
   alembic upgrade head
   # run your test suite
   alembic downgrade -1
   alembic upgrade head     ← round-trip test
         ↓
5. Apply to production during low-traffic window:
   alembic upgrade head
```

## Common patterns

### Adding a nullable column (safe, no downtime)

```python
# In migration file
from alembic import op
import sqlalchemy as sa

def upgrade():
    op.add_column("signals",
        sa.Column("llm_score", sa.Numeric(5, 4), nullable=True)
    )

def downgrade():
    op.drop_column("signals", "llm_score")
```

### Adding a NOT NULL column (needs default or two-step)

```python
# WRONG: will fail if table has existing rows
op.add_column("trades", sa.Column("leverage", sa.SmallInteger(), nullable=False))

# RIGHT: add nullable, backfill, then add NOT NULL constraint
def upgrade():
    # Step 1: add nullable
    op.add_column("trades",
        sa.Column("leverage", sa.SmallInteger(), nullable=True)
    )
    # Step 2: backfill (use a reasonable default)
    op.execute("UPDATE trades SET leverage = 5 WHERE leverage IS NULL")
    # Step 3: add constraint
    op.alter_column("trades", "leverage", nullable=False)

def downgrade():
    op.drop_column("trades", "leverage")
```

### Adding a PostgreSQL enum value (irreversible)

```python
def upgrade():
    # PostgreSQL enums cannot remove values — adding is safe, removing is not
    op.execute("ALTER TYPE trade_outcome ADD VALUE IF NOT EXISTS 'partially_closed'")

def downgrade():
    # Cannot remove enum values in PostgreSQL without recreating the type
    # Document this as a known limitation
    pass
```

### Adding an index without locking the table

```python
def upgrade():
    # CONCURRENTLY avoids table lock — safe for production
    # Cannot run inside a transaction, so use op.execute directly
    op.execute("""
        CREATE INDEX CONCURRENTLY IF NOT EXISTS idx_trades_strategy
        ON trades USING BTREE (account_id, strategy, opened_at DESC)
    """)

def downgrade():
    op.execute("DROP INDEX CONCURRENTLY IF EXISTS idx_trades_strategy")
```

### Data migration pattern

```python
from sqlalchemy.orm import Session

def upgrade():
    # Schema change
    op.add_column("trades",
        sa.Column("r_multiple", sa.Numeric(10, 4), nullable=True)
    )
    # Data backfill using batch update
    op.execute("""
        UPDATE trades
        SET r_multiple = net_pnl / NULLIF(initial_risk, 0)
        WHERE net_pnl IS NOT NULL
          AND initial_risk IS NOT NULL
          AND initial_risk > 0
    """)
```

### Startup migration check

```python
# main.py — run at API server startup
async def run_pending_migrations():
    """Apply pending migrations automatically on startup (dev/staging only)."""
    import subprocess
    result = subprocess.run(
        ["alembic", "upgrade", "head"],
        capture_output=True, text=True
    )
    if result.returncode != 0:
        raise RuntimeError(f"Migration failed:\n{result.stderr}")

# Production: run alembic upgrade head in your CI/CD deploy step,
# not at application startup (startup migration = longer downtime window).
```
